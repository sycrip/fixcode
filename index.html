let currentUser = null; let activeReviewId = null; // Login Handler document.getElementById('loginForm').addEventListener('submit', function(e) { e.preventDefault(); const errDiv = document.getElementById('loginErrorMsg'); errDiv.classList.add('hidden'); const inputUsername = document.getElementById('usernameInput').value.trim(); const inputPassword = document.getElementById('passwordInput').value.trim(); // Case-insensitive account lookup const accountKey = inputUsername.toLowerCase(); const account = ACCOUNTS[accountKey]; if (account && account.pass === inputPassword) { currentUser = { username: account.displayName, role: account.role, canMark: account.canMark }; document.getElementById('currentUserSpan').innerText = account.displayName; document.getElementById('roleBadge').innerText = account.role; document.getElementById('loginOverlay').classList.add('hidden'); document.getElementById('appContent').classList.remove('hidden'); if (currentUser.canMark) { document.getElementById('addCommentSection').classList.remove('hidden'); } else { document.getElementById('addCommentSection').classList.add('hidden'); } renderReviewsList(); } else { errDiv.innerText = "Invalid username or password."; errDiv.classList.remove('hidden'); } }); function logout() { currentUser = null; document.getElementById('loginForm').reset(); document.getElementById('loginOverlay').classList.remove('hidden'); document.getElementById('appContent').classList.add('hidden'); } function showView(viewId) { document.getElementById('menuView').classList.add('hidden'); document.getElementById('detailView').classList.add('hidden'); document.getElementById(viewId).classList.remove('hidden'); } // Data Storage Management function getStoredReviews() { return JSON.parse(localStorage.getItem('cpp_reviews') || '[]'); } function saveReviews(reviews) { localStorage.setItem('cpp_reviews', JSON.stringify(reviews)); } document.getElementById('submissionForm').addEventListener('submit', function(e) { e.preventDefault(); const title = document.getElementById('titleInput').value.trim(); const code = document.getElementById('codeInput').value.trim(); const newReview = { id: Date.now(), title, code, author: currentUser.username, comments: [] }; const reviews = getStoredReviews(); reviews.push(newReview); saveReviews(reviews); this.reset(); renderReviewsList(); }); function renderReviewsList() { const container = document.getElementById('reviewsList'); const reviews = getStoredReviews(); if (reviews.length === 0) { container.innerHTML = '

No code submissions found yet.

'; return; } container.innerHTML = reviews.map(r => `
${escapeHtml(r.title)}
By: ${escapeHtml(r.author)} | Comments: ${r.comments.length}
`).join(''); } function openReview(id) { const reviews = getStoredReviews(); const review = reviews.find(r => r.id === id); if (!review) return; activeReviewId = id; document.getElementById('detailTitle').innerText = review.title; document.getElementById('detailAuthor').innerText = review.author; document.getElementById('detailCode').innerText = review.code; renderComments(review.comments); showView('detailView'); } function renderComments(comments) { const container = document.getElementById('commentsContainer'); if (!comments || comments.length === 0) { container.innerHTML = '

No feedback added yet.

'; return; } container.innerHTML = comments.map(c => `
${escapeHtml(c.author)}
${escapeHtml(c.text)}
`).join(''); } function addComment() { if (!currentUser.canMark) return; const text = document.getElementById('commentInput').value.trim(); if (!text) return; const reviews = getStoredReviews(); const review = reviews.find(r => r.id === activeReviewId); if (review) { review.comments.push({ author: currentUser.username, text }); saveReviews(reviews); renderComments(review.comments); document.getElementById('commentInput').value = ''; } } function escapeHtml(str) { return str.replace(/&/g, "&").replace(//g, ">"); }
